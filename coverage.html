// GenPass defines the interface for handling password generation requests.
type GenPass interface {
        GeneratePassword(c *gin.Context)
}

// genPassHandler is the implementation of the GenPass interface.
type genPassHandler struct {
        genPassSvc service.GenPass
        cfg        config.Config
        cfgStatus  config.ConfigStatus
        cfgError   config.ConfigError
}

// NewGenPass creates a new instance of the genPassHandler with the provided GenPass service and configuration.
func NewGenPass(genPassSvc service.GenPass, cfg config.Config, cfgStatus config.ConfigStatus, cfgError config.ConfigError) GenPass {
        return &genPassHandler{
                genPassSvc: genPassSvc,
                cfg:        cfg,
                cfgStatus:  cfgStatus,
                cfgError:   cfgError,
        }
}

// @Summary Generate Random Password
// @Description GeneratePassword handles the HTTP request for generating a password.
// @Tags Password
// @Produces application/json
// @Success 200 {object} map[string]string
// @Failure 500 {object} map[string]string
// @Router /genpass [get]
func (self *genPassHandler) GeneratePassword(c *gin.Context) {
        pwd, err := self.genPassSvc.GeneratePassword()
        if err != nil || len(pwd) != self.cfg.PasswordLengthDefault {
                c.AbortWithStatusJSON(http.StatusInternalServerError, gin.H{self.cfgStatus.StatusOK: self.cfgError.ErrorInternalServer})
                return
        }
        c.JSON(http.StatusOK, gin.H{self.cfg.PasswordTitle: pwd})
}

type ShortenURL interface {

// shortenURLHandler handles the HTTP requests for shortening URLs.
type shortenURLHandler struct {
        shortenURLService service.ShortenURL
        cfg               config.Config
        cfgError          config.ConfigError
        cfgParameter      config.ConfigParameter
        client            *redis.Client
}

// NewShortenURLHandler creates a new instance of the ShortenURLHandler with the provided service.
func NewShortenURLHandler(svc service.ShortenURL, cfg config.Config, cfgError config.ConfigError, cfgParameter config.ConfigParameter, client *redis.Client) *shortenURLHandler {

                shortenURLService: svc,
                cfg:               cfg,
                cfgError:          cfgError,
                cfgParameter:      cfgParameter,
                client:            client,
        }
}

// CreateShortenLink handles the creation of a shortened URL.
// @Summary Create Shortened URL
// @Description CreateShortenLink handles the HTTP request for creating a shortened URL.
// @Tags link
// @Accept application/json
// @Produce application/json
// @Param input body model.ShortenLinkInput true "Input required"
// @Success 200 {object} map[string]string
// @Failure 400 {object} map[string]string
// @Failure 500 {object} map[string]string
// @Router /v1/links/shorten [post]
func (self *shortenURLHandler) CreateShortenLink(ctx *gin.Context) {
        // Implementation for creating a shortened link
        if err := self.client.Ping(ctx).Err(); err != nil {
                ctx.AbortWithStatusJSON(http.StatusInternalServerError, gin.H{self.cfgError.ErrorTitle: self.cfgError.ErrorRedisConnection})
                return
        }
        input := &model.ShortenLinkInput{}
        if err := ctx.ShouldBindJSON(input); err != nil {
                ctx.AbortWithStatusJSON(http.StatusBadRequest, gin.H{self.cfgError.ErrorTitle: self.cfgError.ErrorInvalidInput})
                return
        }
        shortenLinkOutput, err := self.shortenURLService.CreateShortenLink(ctx, input.Url)
        if err != nil {
                ctx.AbortWithStatusJSON(http.StatusInternalServerError, gin.H{self.cfgError.ErrorTitle: self.cfgError.ErrorInternalServer})
                return
        }
        ctx.JSON(http.StatusOK, shortenLinkOutput)
}

// RedirectToOriginalURL handles the redirection to the original URL based on the provided key code.
// @Summary Redirect to Original URL
// @Description RedirectToOriginalURL handles the HTTP request for redirecting to the original URL based on the provided key code.
// @Tags link
// @Produce application/json
// @Param code path string true "Key Code"
// @Success 302 {string} string "Redirects to the original URL"
// @Failure 400 {object} map[string]string
// @Failure 404 {object} map[string]string
// @Failure 500 {object} map[string]string
// @Router /v1/links/redirect/{code} [get]
func (self *shortenURLHandler) RedirectToOriginalURL(ctx *gin.Context) {
        keyCode := ctx.Param(self.cfgParameter.Code)
        if keyCode == self.cfg.StringEmpty {
                ctx.AbortWithStatusJSON(http.StatusBadRequest, gin.H{self.cfgError.ErrorTitle: self.cfgError.ErrorInvalidCode})
                return
        }
        url, err := self.shortenURLService.GetUrlByCode(ctx, keyCode)
        if err != nil {
                ctx.AbortWithStatusJSON(http.StatusInternalServerError, gin.H{self.cfgError.ErrorTitle: self.cfgError.ErrorInternalServer})
                return
        }
        if url == self.cfg.StringEmpty {
                ctx.AbortWithStatusJSON(http.StatusNotFound, gin.H{self.cfgError.ErrorTitle: self.cfgError.ErrorUrlNotFound})
                return
        }
        ctx.Redirect(http.StatusMovedPermanently, url)
}

type URLStorage interface {
        StoreURL(ctx context.Context, key, url string) error
        GetURL(ctx context.Context, key string) (string, error)
}

type urlStorage struct {
        client *redis.Client
        cfg    *config.Config
}

func NewURLStorage(client *redis.Client, cfg *config.Config) URLStorage {
        return &urlStorage{
                client: client,
                cfg:    cfg,
        }
}

func (self *urlStorage) StoreURL(ctx context.Context, key, url string) error {
        expTime := time.Duration(self.cfg.RedisExpiredTime) * time.Second
        return self.client.Set(ctx, key, url, expTime).Err()
}

func (self *urlStorage) GetURL(ctx context.Context, key string) (string, error) {
        return self.client.Get(ctx, key).Result()
}

// GenPass defines the interface for generating passwords.
type GenPass interface {
        GenerateRandomString(length int) (string, error)
        GeneratePassword() (string, error)
}

// genPassService is the implementation of the GenPass interface.
type genPassService struct {
        cfg      config.Config
        cfgError config.ConfigError
}

// NewGenPass creates a new instance of the genPassService with the provided configuration.
func NewGenPass(cfg config.Config, cfgError config.ConfigError) GenPass {
        return &genPassService{
                cfg:      cfg,
                cfgError: cfgError,
        }
}

func (self *genPassService) GenerateRandomString(length int) (string, error) {
        if length <= 0 {
                log.Error().Msg(self.cfgError.ErrorLengthZero)
                return self.cfg.StringEmpty, errors.New(self.cfgError.ErrorLengthZero)
        }

        result := strings.ReplaceAll(uuid.New().String(), self.cfg.Dash, self.cfg.StringEmpty)
        pass_len := len(result)

        for pass_len < length {
                result += strings.ReplaceAll(uuid.New().String(), self.cfg.Dash, self.cfg.StringEmpty)
                pass_len = len(result)
        }

        result = result[:length]
        return result, nil
}

// GeneratePassword generates a random password using UUIDs and returns it as a string.
func (self *genPassService) GeneratePassword() (string, error) {
        result, err := self.GenerateRandomString(self.cfg.PasswordLengthDefault)
        if err != nil {
                log.Error().Msgf("Error generating random string: %v", err)
                return self.cfg.StringEmpty, err
        }
        return result, nil
}

import (

        "github.com/pvlearning002/ebvn_assignment/internal/repository"

// ShortenURL defines the interface for shortening URLs.
type ShortenURL interface {
        CreateShortenLink(ctx context.Context, url string) (*model.ShortenLinkOutput, error)
        GetUrlByCode(ctx context.Context, keyCode string) (string, error)
}

// shortenURLService is the implementation of the ShortenURLService interface.
type shortenURLService struct {
        urlStorage repository.URLStorage
        genPass    GenPass
        cfg        config.Config
        cfgError   config.ConfigError
}

// NewShortenURLService creates a new instance of the shortenURLService with the provided URL storage and configuration.
func NewShortenURLService(urlStorage repository.URLStorage, genPass GenPass, cfg config.Config, cfgError config.ConfigError) ShortenURL {
        return &shortenURLService{
                urlStorage: urlStorage,
                genPass:    genPass,
                cfg:        cfg,
                cfgError:   cfgError,
        }
}

func (self *shortenURLService) generateKey(ctx context.Context) (string, error) {
        key, err := self.genPass.GenerateRandomString(self.cfg.KeycodeLengthDefault)
        if err != nil {
                return self.cfg.StringEmpty, err
        }
        url_check, _ := self.urlStorage.GetURL(ctx, key)
        for url_check != self.cfg.StringEmpty {
                key, err = self.genPass.GenerateRandomString(self.cfg.KeycodeLengthDefault)
                if err != nil {
                        return self.cfg.StringEmpty, err
                }
                url_check, _ = self.urlStorage.GetURL(ctx, key)
        }

        return key, nil
}

// ShortenURL shortens the provided original URL and returns the shortened URL.
func (self *shortenURLService) CreateShortenLink(ctx context.Context, url string) (*model.ShortenLinkOutput, error) {
        key, err := self.generateKey(ctx)
        if err != nil {
                return &model.ShortenLinkOutput{KeyCode: self.cfg.StringEmpty, Message: err.Error()}, err
        }

        err = self.urlStorage.StoreURL(ctx, key, url)
        if err != nil {
                return &model.ShortenLinkOutput{KeyCode: self.cfg.StringEmpty, Message: err.Error()}, err
        }
        return &model.ShortenLinkOutput{KeyCode: key, Message: "Shorten URL generated successfully"}, nil
}

// ShortenURL shortens the provided original URL and returns the shortened URL.
func (self *shortenURLService) GetUrlByCode(ctx context.Context, keyCode string) (string, error) {
        url, err := self.urlStorage.GetURL(ctx, keyCode)
        if err != nil {
                return self.cfg.StringEmpty, err
        }
        return url, nil
}

import (

var dockerInspect = func() ([]byte, error) {
        cmd := exec.Command(
                "docker",
                "inspect",
                "-f",
                "{{.State.Running}}",
                "redis",
        )
        return cmd.Output()
}

// NewRedisClient creates a new Redis client using the provided environment prefix for configuration.
func NewRedisClient(envPrefix string) (*redis.Client, error) {
        cfg, err := newConfig(envPrefix)
        if err != nil {
                return nil, err
        }

        rdb := redis.NewClient(&redis.Options{
                Addr:     cfg.Address,
                Password: cfg.Password,
                DB:       cfg.DB,
        })

        return rdb, nil
}

func CheckRedisConnection(
        ctx context.Context,
        client *redis.Client,
) error {
        ctx, cancel := context.WithTimeout(ctx, 3*time.Second)
        defer cancel()

        if err := client.Ping(ctx).Err(); err != nil {
                return fmt.Errorf("redis connection failed: %w", err)
        }

        return nil
}

func IsRedisContainerRunning() bool {
        output, err := dockerInspect()
        if err != nil {
                return false
        }

        return strings.TrimSpace(string(output)) == "true"
}
