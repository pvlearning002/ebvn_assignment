        genPassSvc service.GenPass
        cfg        config.Config
        cfgError   config.ConfigError
        logger     utils.LoggerUtils
}

// NewGenPass creates a new instance of the genPassHandler with the provided GenPass service and configuration.

// @Description GeneratePassword handles the HTTP request for generating a password.
// @Tags Password
// @Produces application/json
// @Success 200 {object} map[string]string
// @Failure 500 {object} map[string]string
// @Router /genpass [get]

        pwd, err := self.genPassSvc.GeneratePassword()

                return
        }
        c.JSON(http.StatusOK, gin.H{self.cfg.PasswordTitle: pwd})
}

// CreateShortenLink handles the creation of a shortened URL.
// @Summary Create Shortened URL
// @Description CreateShortenLink handles the HTTP request for creating a shortened URL.
// @Tags link
// @Accept application/json
// @Produce application/json
// @Param input body model.ShortenLinkInput true "Input required"
// @Success 200 {object} map[string]string
// @Failure 400 {object} map[string]string
// @Failure 500 {object} map[string]string
// @Router /v1/links/shorten [post]
func (self *shortenURLHandler) CreateShortenLink(ctx *gin.Context) {
        input := &model.ShortenLinkInput{}
        if err := ctx.ShouldBindJSON(input); err != nil {
                self.logger.WriteLogError(err.Error())
                ctx.AbortWithStatusJSON(http.StatusBadRequest, gin.H{self.cfgError.ErrorTitle: self.cfgError.ErrorInvalidInput})
                return
        }
        shortenLinkOutput, err := self.shortenURLService.CreateShortenLink(ctx, input.Url)
        if err != nil {
                self.logger.WriteLogError(err.Error())
                ctx.AbortWithStatusJSON(http.StatusInternalServerError, gin.H{self.cfgError.ErrorTitle: self.cfgError.ErrorInternalServer})
                return
        }
        ctx.JSON(http.StatusOK, shortenLinkOutput)
}

// RedirectToOriginalURL handles the redirection to the original URL based on the provided key code.
// @Summary Redirect to Original URL
// @Description RedirectToOriginalURL handles the HTTP request for redirecting to the original URL based on the provided key code.
// @Tags link
// @Produce application/json
// @Param code path string true "Key Code"
// @Success 302 {string} string "Redirects to the original URL"
// @Failure 400 {object} map[string]string
// @Failure 404 {object} map[string]string
// @Failure 500 {object} map[string]string
// @Router /v1/links/redirect/{code} [get]
func (self *shortenURLHandler) RedirectToOriginalURL(ctx *gin.Context) {
        keyCode := ctx.Param(self.cfgParameter.Code)
        if keyCode == self.cfg.StringEmpty {
                self.logger.WriteLogError(self.cfgError.ErrorInvalidCode)
                ctx.AbortWithStatusJSON(http.StatusBadRequest, gin.H{self.cfgError.ErrorTitle: self.cfgError.ErrorInvalidCode})
                return
        }
        url, err := self.shortenURLService.GetUrlByCode(ctx, keyCode)
        if err != nil {
                self.logger.WriteLogError(err.Error())
                ctx.AbortWithStatusJSON(http.StatusInternalServerError, gin.H{self.cfgError.ErrorTitle: self.cfgError.ErrorInternalServer})
                return
        }
        if url == self.cfg.StringEmpty {
                ctx.AbortWithStatusJSON(http.StatusNotFound, gin.H{self.cfgError.ErrorTitle: self.cfgError.ErrorUrlNotFound})
                return
        }
        ctx.Redirect(http.StatusMovedPermanently, url)
}

// ShortenURL defines the interface for shortening URLs.
type ShortenURL interface {
        CreateShortenLink(ctx context.Context, url string) (*model.ShortenLinkOutput, error)
        GetUrlByCode(ctx context.Context, keyCode string) (string, error)
}

// shortenURLService is the implementation of the ShortenURLService interface.
type shortenURLService struct {
        urlStorage  repository.URLStorage
        cfg         config.Config
        logger      utils.LoggerUtils
        stringUtils utils.StringUtils
}

// NewShortenURLService creates a new instance of the shortenURLService with the provided URL storage and configuration.
func NewShortenURLService(urlStorage repository.URLStorage, cfg config.Config, logger utils.LoggerUtils, stringUtils utils.StringUtils) ShortenURL {
        return &shortenURLService{
                urlStorage:  urlStorage,
                cfg:         cfg,
                logger:      logger,
                stringUtils: stringUtils,
        }
}

func (self *shortenURLService) generateKey(ctx context.Context) (string, error) {
        key, err := self.stringUtils.GenerateRandomString(self.cfg.KeycodeLengthDefault)
        if err != nil {
                self.logger.WriteLogError(err.Error())
                return self.cfg.StringEmpty, err
        }
        url_check, _ := self.urlStorage.GetURL(ctx, key)
        for url_check != self.cfg.StringEmpty {
                key, err = self.stringUtils.GenerateRandomString(self.cfg.KeycodeLengthDefault)
                if err != nil {
                        self.logger.WriteLogError(err.Error())
                        return self.cfg.StringEmpty, err
                }
                url_check, _ = self.urlStorage.GetURL(ctx, key)
        }

        return key, nil
}

// ShortenURL shortens the provided original URL and returns the shortened URL.
func (self *shortenURLService) CreateShortenLink(ctx context.Context, url string) (*model.ShortenLinkOutput, error) {
        key, err := self.generateKey(ctx)
        if err != nil {
                self.logger.WriteLogError(err.Error())
                return &model.ShortenLinkOutput{KeyCode: self.cfg.StringEmpty, Message: err.Error()}, err
        }

        err = self.urlStorage.StoreURL(ctx, key, url)
        if err != nil {
                self.logger.WriteLogError(err.Error())
                return &model.ShortenLinkOutput{KeyCode: self.cfg.StringEmpty, Message: err.Error()}, err
        }
        return &model.ShortenLinkOutput{KeyCode: key, Message: "Shorten URL generated successfully"}, nil
}

// ShortenURL shortens the provided original URL and returns the shortened URL.

func (self *shortenURLService) GetUrlByCode(
        ctx context.Context,
        keyCode string,
) (string, error) {

        url, err := self.urlStorage.GetURL(ctx, keyCode)

        if errors.Is(err, redis.Nil) {
                self.logger.WriteLogError(err.Error())
                return self.cfg.StringEmpty, nil
        }

        if err != nil {
                self.logger.WriteLogError(err.Error())
                return self.cfg.StringEmpty, err
        }

        return url, nil
}

type LoggerUtils interface {
        SetLogLevel(levelStr string)
        GetPathFunction(skip ...int) string
        BuildLogMessage(message string, skip ...int) string
        WriteLogInfo(message string, skip ...int)
        WriteLogError(message string, skip ...int)
        WriteLogDebug(message string, skip ...int)
        WriteLogWarning(message string, skip ...int)
}

type loggerUtils struct {
}

func NewLoggerUtils() LoggerUtils {
        return &loggerUtils{}
}

func (self *loggerUtils) SetLogLevel(levelStr string) {
        level := zerolog.NoLevel
        level, err := zerolog.ParseLevel(levelStr)
        if err != nil {
                level = zerolog.InfoLevel
        }
        zerolog.SetGlobalLevel(level)
}

func (self *loggerUtils) GetPathFunction(skip ...int) string {
        default_skip := 3
        if len(skip) > 0 {
                default_skip = skip[0]
        }
        pc, _, _, ok := runtime.Caller(default_skip)
        result := "unknown"
        if ok {
                fn := runtime.FuncForPC(pc)
                if fn != nil {
                        result = fn.Name()
                }
        }
        return result
}

// Added more function path into log
func (self *loggerUtils) BuildLogMessage(message string, skip ...int) string {
        caller := self.GetPathFunction(skip...)
        output := "func: " + caller + " - log: " + message
        return output
}

func (self *loggerUtils) WriteLogInfo(message string, skip ...int) {
        log.Info().Msg(self.BuildLogMessage(message, skip...))
}

func (self *loggerUtils) WriteLogError(message string, skip ...int) {
        log.Error().Msg(self.BuildLogMessage(message, skip...))
}

func (self *loggerUtils) WriteLogDebug(message string, skip ...int) {
        log.Debug().Msg(self.BuildLogMessage(message, skip...))
}

func (self *loggerUtils) WriteLogWarning(message string, skip ...int) {
        log.Warn().Msg(self.BuildLogMessage(message, skip...))
}

type RedisClientUtils interface {
        NewRedisClient(cfg config.ConfigRedis) (*redis.Client, error)
        DockerInspect() ([]byte, error)
        IsRedisConnect(ctx context.Context) (bool, error)
        IsRedisContainerRunning() bool
        StoreStringValue(ctx context.Context, key, stringValue string) error
        GetStringValue(ctx context.Context, key string) (string, error)
}

type redisClientUtils struct {
        client          *redis.Client
        cfg             config.Config
        cfgRedis        config.ConfigRedis
        cfgError        config.ConfigError
        dockerInspectFn func() ([]byte, error)
}

func NewRedisClientUtils(
        cfg config.Config,
        cfgRedis config.ConfigRedis,
        cfgError config.ConfigError) RedisClientUtils {
        client := redis.NewClient(&redis.Options{
                Addr:     cfgRedis.Address,
                Password: cfgRedis.Password,
                DB:       cfgRedis.DB,
        })
        return &redisClientUtils{
                client:   client,
                cfg:      cfg,
                cfgRedis: cfgRedis,
                cfgError: cfgError,
        }
}

// NewRedisClient creates a new Redis client using the provided environment prefix for configuration.
func (self *redisClientUtils) NewRedisClient(cfg config.ConfigRedis) (*redis.Client, error) {
        rdb := redis.NewClient(&redis.Options{
                Addr:     cfg.Address,
                Password: cfg.Password,
                DB:       cfg.DB,
        })

        return rdb, nil
}

func (self *redisClientUtils) DockerInspect() ([]byte, error) {
        if self.dockerInspectFn != nil {
                return self.dockerInspectFn()
        }
        cmd := exec.Command(
                "docker",
                "inspect",
                "-f",
                "{{.State.Running}}",
                "redis",
        )
        return cmd.Output()

}

func (self *redisClientUtils) IsRedisConnect(
        ctx context.Context,
) (bool, error) {
        ctx, cancel := context.WithTimeout(ctx, 3*time.Second)
        defer cancel()

        if err := self.client.Ping(ctx).Err(); err != nil {
                return false, fmt.Errorf("redis connection failed: %w", err)
        }

        return true, nil
}

func (self *redisClientUtils) IsRedisContainerRunning() bool {
        output, err := self.DockerInspect()
        if err != nil {
                return false
        }
        return strings.TrimSpace(string(output)) == "true"
}

func (self *redisClientUtils) StoreStringValue(ctx context.Context, key, stringValue string) error {
        if isConnected, err := self.IsRedisConnect(ctx); err == nil && isConnected {
                expTime := time.Duration(self.cfg.RedisExpiredTime) * time.Second
                return self.client.Set(ctx, key, stringValue, expTime).Err()
        }
        return errors.New(self.cfgError.ErrorRedisConnection)
}

func (self *redisClientUtils) GetStringValue(ctx context.Context, key string) (string, error) {
        if isConnected, err := self.IsRedisConnect(ctx); err == nil && isConnected {
                return self.client.Get(ctx, key).Result()
        }
        return self.cfg.StringEmpty, errors.New(self.cfgError.ErrorRedisConnection)

}

                return ResponseMessage{
                        Message: self.cfgError.ErrorInternalServer,

type StringUtils interface {
        GenerateRandomString(length int) (string, error)
}

type stringUtils struct {
        cfg      config.Config
        cfgError config.ConfigError
        logger   LoggerUtils
}

func NewStringUtils(cfg config.Config, cfgError config.ConfigError, logger LoggerUtils) StringUtils {
        return &stringUtils{
                cfg:      cfg,
                cfgError: cfgError,
                logger:   logger,
        }
}

func (self *stringUtils) GenerateRandomString(length int) (string, error) {
        if length <= 0 {
                self.logger.WriteLogError(self.cfgError.ErrorLengthZero)
                return self.cfg.StringEmpty, errors.New(self.cfgError.ErrorLengthZero)
        }

        result := strings.ReplaceAll(uuid.New().String(), self.cfg.Dash, self.cfg.StringEmpty)
        pass_len := len(result)

        for pass_len < length {
                result += strings.ReplaceAll(uuid.New().String(), self.cfg.Dash, self.cfg.StringEmpty)
                pass_len = len(result)
        }

        result = result[:length]
        return result, nil
}
